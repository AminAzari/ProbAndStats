\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  فرض صفر و فرض مقابل را برای مسایل مهندسی صورت‌بندی کنند
\item
  خطاهای نوع اول و نوع دوم و پیامدهای مهندسی آن‌ها را تشخیص دهند
\item
  آماره آزمون و مقادیر \lr{$p$} را محاسبه کنند
\item
  آزمون‌های \lr{$Z$} و \lr{$t$} را برای میانگین‌ها انجام دهند
\item
  مقادیر \lr{$p$} را به‌درستی تفسیر کنند (و از تصورهای نادرست رایج پرهیز کنند)
\item
  میان آزمون‌های یک‌طرفه و دوطرفه انتخاب کنند
\end{enumerate}

\end{PSbox}

\PSsection{15.1}{مقدمه: تصمیم‌گیری از داده}

\textbf{پرسش مهندسی:} «آیا این حسگر اریب است؟» «آیا فرایند تغییر کرده است؟» «آیا الگوریتم جدید بهتر عمل می‌کند؟»

به یک \textbf{چارچوب نظام‌مند} نیاز داریم تا به این پرسش‌ها از روی داده پاسخ دهیم، در حالی که احتمال خطا را کنترل می‌کنیم.

\PSsection{15.2}{چارچوب}

\PSsubsection{فرض صفر \lr{$H_{0}$}}

\textbf{فرض پیش‌فرض} \lr{—} معمولاً «هیچ چیز تغییر نکرده است» یا «هیچ اثری وجود ندارد»:

\begin{itemize}
\tightlist
\item
  \lr{$H_{0}: \mu = \mu_{0}$} (حسگر اریب نیست)
\item
  \lr{$H_{0}: \mu_{1} = \mu_{2}$} (دو روش به یک اندازه عمل می‌کنند)
\item
  \lr{$H_{0}: \sigma^{2} = \sigma_{0}^{2}$} (واریانس تغییر نکرده است)
\end{itemize}

\PSsubsection{فرض مقابل \lr{$H_{1}$}}

آنچه می‌خواهیم \textbf{آشکار کنیم} یا اثبات کنیم:

\begin{itemize}
\tightlist
\item
  \lr{$H_{1}: \mu \ne \mu_{0}$} (حسگر اریب است \lr{—} دوطرفه)
\item
  \lr{$H_{1}: \mu > \mu_{0}$} (عملکرد بهبود یافته است \lr{—} یک‌طرفه)
\item
  \lr{$H_{1}: \sigma^{2} > \sigma_{0}^{2}$} (واریانس افزایش یافته است)
\end{itemize}

\PSsubsection{تصمیم}

بر اساس داده، یکی از این دو:

\begin{itemize}
\tightlist
\item
  \textbf{رد \lr{$H_{0}$}} \lr{\mbox{\ensuremath{\to}}} شواهد از \lr{$H_{1}$} پشتیبانی می‌کنند
\item
  \textbf{عدم رد \lr{$H_{0}$}} \lr{\mbox{\ensuremath{\to}}} شواهد ناکافی علیه \lr{$H_{0}$}
\end{itemize}

\begin{PSquote}

\lr{\mbox{\PSwarn{}}} «عدم رد \lr{\mbox{$H_{0}$» \ensuremath{\neq} «$H_{0}$}} درست است». نبودِ شواهد \lr{\mbox{\ensuremath{\neq}}} شواهدِ نبود.

\end{PSquote}

\PSsubsection{خطای نوع اول (هشدار کاذب) \lr{—} با احتمال \lr{$\alpha$}}

رد \lr{$H_{0}$} هنگامی که در واقع درست است. «فریاد گرگ.»

قیاس مهندسی: اعلام تغییر یک فرایند در حالی که تغییر نکرده است \lr{\mbox{\ensuremath{\to}}} کالیبراسیون مجدد بی‌مورد.

\PSsubsection{خطای نوع دوم (آشکارسازی از‌دست‌رفته) \lr{—} با احتمال \lr{$\beta$}}

عدم رد \lr{$H_{0}$} هنگامی که \lr{$H_{1}$} در واقع درست است. «از دست دادن سیگنال.»

قیاس مهندسی: عدم آشکارسازی اریبی یک حسگر \lr{\mbox{\ensuremath{\to}}} ادامه اندازه‌گیری‌های اریب.

\PSsubsection{توازن}

کاهش \lr{$\alpha$} (هشدارهای کاذب کمتر) \lr{\mbox{\ensuremath{\to}}} افزایش \lr{$\beta$} (آشکارسازی‌های از‌دست‌رفته بیشتر)، و برعکس.

\lr{$\alpha$} را ثابت کنید (سطح معناداری)، سپس توان \lr{$= 1 - \beta$} را بیشینه کنید.

\PSsection{15.4}{سطح معناداری \lr{$\alpha$}}

\textbf{حداکثر نرخ قابل‌قبول خطای نوع اول}، که \textbf{پیش از} نگاه‌کردن به داده انتخاب می‌شود.

انتخاب‌های رایج: \lr{$\alpha = 0.05$}، \PSnum{0.01}، \PSnum{0.10}

\textbf{انتخاب \lr{$\alpha$}:} هزینه هشدار کاذب را در نظر بگیرید:

\begin{itemize}
\tightlist
\item
  \lr{$\alpha = 0.01$}: بسیار محافظه‌کارانه (مثلاً ایمنی تجهیزات پزشکی)
\item
  \lr{$\alpha = 0.05$}: استاندارد در بیشتر شاخه‌های مهندسی/علوم
\item
  \lr{$\alpha = 0.10$}: با رواداری بیشتر (تحلیل اکتشافی)
\end{itemize}

\PSsection{15.5}{توان یک آزمون}

\PSdisplay{\PSmtext{Power} = 1 - \beta = P(\PSmtext{reject } H_0 | H_1 \PSmtext{ is true})}

توان به موارد زیر بستگی دارد:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{اندازه اثر:} تفاوت واقعی بزرگ‌تر \lr{\mbox{\ensuremath{\to}}} آشکارسازی آسان‌تر \lr{\mbox{\ensuremath{\to}}} توان بیشتر
\item
  \textbf{اندازه نمونه \lr{$n$}:} داده بیشتر \lr{\mbox{\ensuremath{\to}}} توان بیشتر
\item
  \textbf{سطح معناداری \lr{$\alpha$}:} \lr{$\alpha$} بزرگ‌تر \lr{\mbox{\ensuremath{\to}}} توان بیشتر (اما هشدار کاذب بیشتر)
\item
  \textbf{تغییرپذیری \lr{$\sigma$}:} نویز کمتر \lr{\mbox{\ensuremath{\to}}} توان بیشتر
\end{enumerate}

\PSsection{15.6}{مقدار \lr{$p$}}

\begin{PSbox}[unbreakable]{def}{تعریف}

\textbf{مقدار \lr{$p$}} احتمال به‌دست‌آوردن یک آماره آزمون \textbf{به‌اندازه مشاهده‌شده حدی یا حدی‌تر} است، \textbf{با فرض درست‌بودن \lr{$H_{0}$}}.

\PSdisplay{p\PSmtext{-value} = P(\PSmtext{data this extreme or more} | H_0 \PSmtext{ true})}

\end{PSbox}

\PSsubsection{قاعده تصمیم}

\begin{itemize}
\tightlist
\item
  اگر مقدار \lr{$p < \alpha \to H_{0}$} را رد کنید
\item
  اگر مقدار \lr{$p \ge \alpha \to H_{0}$} را رد نکنید
\end{itemize}

\PSsubsection{\lr{\mbox{\PSyes{}}} تفسیر درست}

\begin{PSquote}

«مقدار \lr{$p$} احتمال مشاهده داده‌ای به این حدی است (یا حدی‌تر) اگر \lr{$H_{0}$} درست بود. مقدار \lr{$p$} کوچک به این معناست که داده تحت \lr{$H_{0}$} نامحتمل است.»

\end{PSquote}

\PSsubsection{\lr{\mbox{\PSno{}}} تفسیر نادرست}

\begin{PSquote}

{«مقدار \lr{$p$} احتمال درست‌بودن \lr{$H_{0}$} است.»}

\end{PSquote}

\textbf{چرا نادرست است:} \lr{$H_{0}$} یا درست است یا نادرست (ثابت). مقدار \lr{$p$} هیچ چیزی درباره \lr{$P(H_{0}\ \PSmtext{true})$} نمی‌گوید. این مقدار می‌سنجد که داده تحت \lr{$H_{0}$} چقدر شگفت‌آور است.

\PSsubsection{درجه‌بندی}

\endlastfoot
\lr{\mbox{\textgreater{} 0.10}} & ضعیف یا هیچ \\

\PSsection{15.7}{آزمون‌های یک‌طرفه در برابر دوطرفه}

\PSsubsection{دوطرفه \lr{\mbox{($H_{1}$: $\mu \ne \mu_{0}$)}}}

هنگامی به کار می‌رود که انحراف در \textbf{هر} جهت مهم باشد.

\begin{itemize}
\tightlist
\item
  رد کنید اگر \lr{$\lvert T\rvert > t_{\alpha /2, n-1}$}
\item
  مقدار \lr{$p = 2 \cdot P(T > \lvert t_{\PSmtext{obs}}\rvert)$}
\end{itemize}

\PSsubsection{یک‌طرفه (\lr{\mbox{$H_{1}$: $\mu > \mu_{0}$}} یا \lr{$H_{1}: \mu < \mu_{0}$})}

هنگامی به کار می‌رود که تنها یک جهت انحراف اهمیت دارد.

\begin{itemize}
\tightlist
\item
  \lr{$H_{1}: \mu > \mu_{0}$}: رد کنید اگر \lr{$T > t_{\alpha, n-1}$}
\item
  \lr{$H_{1}: \mu < \mu_{0}$}: رد کنید اگر \lr{$T < -t_{\alpha, n-1}$}
\end{itemize}

\PSsubsection{چه زمانی کدام را به کار ببریم}

\PSsection{15.8}{آزمون‌های میانگین}

\PSsubsection{آزمون \lr{$Z$} تک‌نمونه‌ای (\lr{$\sigma$} معلوم)}

آزمون: \lr{$H_{0}: \mu = \mu_{0}$}

آماره آزمون: \lr{$Z = \frac{\bar{X} - \mu_{0}}{\frac{\sigma}{\sqrt{n}}}$}

تحت \lr{$H_{0}: Z \sim N(0,\,1)$}

\PSsubsection{آزمون \lr{$t$} تک‌نمونه‌ای (\lr{$\sigma$} نامعلوم)}

آزمون: \lr{$H_{0}: \mu = \mu_{0}$}

آماره آزمون: \lr{$T = \frac{\bar{X} - \mu_{0}}{\frac{S}{\sqrt{n}}}$}

تحت \lr{$H_{0}: T \sim t(n-1)$}

\PSsubsection{آزمون \lr{$t$} دو‌نمونه‌ای (مقایسه دو میانگین)}

آزمون: \lr{$H_{0}: \mu_{1} = \mu_{2}$}

آماره آزمون: \lr{$T = \frac{\bar{X}_{1} - \bar{X}_{2}}{\sqrt{\frac{S_{1}^{2}}{n_{1}} + \frac{S_{2}^{2}}{n_{2}}}}$}

درجات آزادی تقریبی بر اساس فرمول ولچ-ساترث‌ویت.

\PSsection{15.9}{آزمون واریانس (آزمون کای‌دو)}

\PSsubsection{آزمون تک‌نمونه‌ای}

\lr{$H_{0}: \sigma^{2} = \sigma_{0}^{2}$}

آماره آزمون: \lr{$\chi^{2} = \frac{(n-1)S^{2}}{\sigma_{0}^{2}}$}

تحت \lr{$H_{0}: \chi^{2} \sim \chi^{2}(n-1)$}

\PSsubsection{تصمیم}

\begin{itemize}
\tightlist
\item
  \lr{$H_{1}: \sigma^{2} > \sigma_{0}^{2}$}: رد کنید اگر \lr{$\chi^{2} > \chi^{2}_{\alpha, n-1}$}
\item
  \lr{$H_{1}: \sigma^{2} \ne \sigma_{0}^{2}$}: رد کنید اگر \lr{$\chi^{2} < \chi^{2}_{1-\alpha /2, n-1}$} یا \lr{$\chi^{2} > \chi^{2}_{\alpha /2, n-1}$}
\end{itemize}

\PSsection{15.10}{رویه گام‌به‌گام آزمون فرض}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{بیان فرض‌ها:} \lr{$H_{0}$} و \lr{$H_{1}$}
\item
  \textbf{انتخاب سطح معناداری:} \lr{$\alpha$} (مثلاً \PSnum{0.05)}
\item
  \textbf{انتخاب آماره آزمون:} \lr{$Z$}، \lr{$t$} یا \lr{$\chi^{2}$} بسته به سناریو
\item
  \textbf{تعیین مقدار(های) بحرانی} یا محاسبه مقدار \lr{$p$}
\item
  \textbf{محاسبه آماره آزمون} از داده
\item
  \textbf{اتخاذ تصمیم:} رد \lr{$H_{0}$} اگر آماره آزمون در ناحیه بحرانی باشد (یا \lr{$p < \alpha$})
\item
  \textbf{بیان نتیجه‌گیری} به زبان مهندسی
\end{enumerate}

\PSsection{15.11}{مثال‌های مهندسی}

\begin{PSbox}[unbreakable]{ex}{مثال 1: آیا یک حسگر اریب است؟}

یک حسگر باید بدون ورودی مقدار \lr{\mbox{0V}} را بخواند. 20 اندازه‌گیری \lr{$\bar{X} = 0.012\,\mathrm{V}$} و \lr{$S = 0.03\,\mathrm{V}$} می‌دهد.

\lr{$H_{0}: \mu = 0$} (بدون اریبی)، \lr{$H_{1}: \mu \ne 0$} (اریب)، \lr{$\alpha = 0.05$}

\lr{$\displaystyle T = \frac{0.012 - 0}{\frac{0.03}{\sqrt{20}}} = \frac{0.012}{0.00671} = 1.789$}

\lr{$t_{0.025, 19} = 2.093$}. از آنجا که \lr{$\lvert T\rvert = 1.789 < 2.093 \to \ast \ast H_{0}$} را رد نکنید.**

مقدار \lr{$p = 2 \cdot P(t(19) > 1.789) \approx 0.090$}. در سطح 5\% معنادار نیست.

نتیجه‌گیری: شواهد کافی برای نتیجه‌گیری اریب‌بودن حسگر وجود ندارد.

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 2: آیا فرایند تولید تغییر کرده است؟}

پیش از تغییر: \lr{$\mu_{0} = 100\,\Omega$} (تثبیت‌شده). پس از اصلاح: \lr{$n = 30$}، \lr{$\bar{X} = 101.2\,\Omega$}، \lr{$S = 3.5\,\Omega$}.

\lr{$H_{0}: \mu = 100$}، \lr{$H_{1}: \mu \ne 100$}، \lr{$\alpha = 0.05$}

\lr{$\displaystyle T = \frac{101.2 - 100}{\frac{3.5}{\sqrt{30}}} = \frac{1.2}{0.639} = 1.878$}

\lr{$t_{0.025, 29} = 2.045$. $\lvert T\rvert = 1.878 < 2.045 \to \ast \ast H_{0}$} را رد نکنید.**

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 3: آیا الگوریتم جدید \lr{\mbox{BER}} را بهبود می‌دهد؟}

الگوریتم قدیمی: \lr{$\mathrm{BER}_{0} = 10^{-3}$}. الگوریتم جدید آزمایش‌شده: \lr{$n = 50000$} بیت، 38 خطا.

\lr{$\displaystyle \mathrm{BER}_{\mathrm{new}} = \frac{38}{50000} = 7.6 \times 10^{-4}$}

\lr{$H_{0}: p = 0.001$}، \lr{$H_{1}: p < 0.001$} (بهبود)، \lr{$\alpha = 0.05$}

\lr{$\displaystyle Z = \frac{\hat{p} - p_{0}}{\sqrt{\frac{p_{0}(1-p_{0})}{n}}} = \frac{0.00076 - 0.001}{\sqrt{0.001 \times \frac{0.999}{50000}}} = -\frac{0.00024}{0.000141} = -1.70$}

مقدار \lr{$p = P(Z < -1.70) = 0.0446 < 0.05 \to \ast \ast H_{0}$} را رد کنید.**

نتیجه‌گیری: شواهد از این که الگوریتم جدید \lr{\mbox{BER}} کمتری دارد پشتیبانی می‌کند.

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 4: آیا توان نویز افزایش یافته است؟}

مقدار تاریخی: \lr{$\sigma_{0}^{2} = 0.04\,\mathrm{V}^{2}$}. اندازه‌گیری‌های جدید: \lr{$n = 25$}، \lr{$S^{2} = 0.058\,\mathrm{V}^{2}$}.

\lr{$H_{0}: \sigma^{2} = 0.04$}، \lr{$H_{1}: \sigma^{2} > 0.04$}، \lr{$\alpha = 0.05$}

\lr{$\displaystyle \chi^{2} = \frac{(24)(0.058)}{0.04} = 34.8$}

\lr{$\chi^{2}_{0.05, 24} = 36.415$}. از آنجا که \lr{$34.8 < 36.415 \to \ast \ast H_{0}$} را رد نکنید.**

\end{PSbox}

\PSsection{15.13}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

یک مشخصه ساخت و تولید میانگین مقاومت = \lr{\mbox{100\ensuremath{\Omega}}} را می‌طلبد. یک نمونه 16 مقاومتی \lr{$\bar{X} = 101.5\,\Omega$} و \lr{$S = 3\,\Omega$} می‌دهد. در \lr{$\alpha = 0.05$}، آیا شاهدی بر رانش فرایند وجود دارد؟

\PSsolution{} \lr{$H_{0}: \mu = 100$}، \lr{$H_{1}: \mu \ne 100$. $T = \frac{101.5-100}{\frac{3}{4}} = 2.0$. $t_{0.025,15} = 2.131$. $\lvert T\rvert = 2.0 < 2.131$ \ensuremath{\to}} رد نکنید. مقدار \lr{$p \approx 0.064 > 0.05$}.

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

میانگین تأخیر بسته \lr{\mbox{5ms}} بود. پس از ارتقای شبکه، 40 اندازه‌گیری \lr{$\bar{X} = 4.6\,\mathrm{ms}$} و \lr{$S = 1.2\,\mathrm{ms}$} می‌دهد. در \lr{$\alpha = 0.01$} آزمون کنید که آیا تأخیر کاهش یافته است.

\PSsolution{} \lr{$H_{0}: \mu = 5$}، \lr{$H_{1}: \mu < 5$. $T = \frac{4.6-5}{\frac{1.2}{\sqrt{40}}} = -\frac{0.4}{0.190} = -2.11$. $t_{0.01,39} \approx -2.426$. $T = -2.11 > -2.426$ \ensuremath{\to}} در سطح 1\% رد نکنید. (در سطح 5\% رد می‌شد چون \lr{$t_{0.05,39} \approx -1.685.$})

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 3}

توضیح دهید چرا «\lr{$p = 0.04$} به این معناست که تنها 4\% شانس وجود دارد که \lr{$H_{0}$} درست باشد» \textbf{نادرست} است.

\PSsolution{} مقدار \lr{$p$} همان \lr{$P(\text{\rl{داده‌ای به این حدی}}\ \mid H_{0}\ \PSmtext{true})$} است، نه \lr{$P(H_{0}\ \PSmtext{true}\ \mid \text{\rl{داده}})$}. این مقدار به \lr{$H_{0}$} درست‌بودن مشروط می‌شود و درباره حدی‌بودن داده می‌پرسد. برای به‌دست‌آوردن \lr{$P(H_{0}\ \PSmtext{true}\ \mid \text{\rl{داده}})$} به تحلیل بیزی با یک پیشین نیاز است. مقدار \lr{$p$} ویژگی فرایند تولید داده تحت \lr{$H_{0}$} است، نه احتمال پسینی درباره \lr{$H_{0}$}.

\end{PSbox}
