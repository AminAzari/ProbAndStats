\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  بازه‌های اطمینان را به‌درستی تفسیر کنند (تفسیر فراوانی‌گرا)
\item
  بازه‌های اطمینان برای میانگین بسازند (واریانس معلوم و نامعلوم)
\item
  توزیع \lr{$t$} را برای نمونه‌های کوچک به کار ببرند
\item
  بازه‌های اطمینان برای واریانس بسازند
\item
  اندازه نمونه لازم را تعیین کنند
\item
  از تصورهای نادرست رایج درباره بازه‌های اطمینان پرهیز کنند
\end{enumerate}

\end{PSbox}

\PSsection{14.1}{مقدمه: چرا برآورد نقطه‌ای کافی نیست}

یک برآورد نقطه‌ای (مثلاً \lr{$\bar{X} = 3.47$}) هیچ اطلاعاتی درباره \textbf{دقت} نمی‌دهد. باید عدم‌قطعیت را کمّی کنیم:

«میانگین واقعی \PSnum{3.47} برآورد می‌شود، \textbf{با تقریبی} به اندازه‌ای.»

یک \textbf{بازه اطمینان} بازه‌ای فراهم می‌کند که با سطح اطمینان معین، پارامتر واقعی را در بر می‌گیرد.

\PSsection{14.2}{تفسیر درست بازه‌های اطمینان}

\PSsubsection{\lr{\mbox{\PSyes{}}} تفسیر درست (فراوانی‌گرا)}

\begin{PSquote}

«اگر این آزمایش را بارها تکرار کنیم و هر بار یک بازه اطمینان 95\% محاسبه کنیم، آنگاه تقریباً 95\% از آن بازه‌ها شامل پارامتر واقعی \lr{$\mu$} خواهند بود.»

\end{PSquote}

این \textbf{رویه} نرخ موفقیت 95\% دارد. هر بازه مشخص یا شامل \lr{$\mu$} است یا نیست.

\PSsubsection{\lr{\mbox{\PSno{}}} تفسیر نادرست}

\begin{PSquote}

{«احتمال 95\% وجود دارد که \lr{$\mu$} داخل این بازه خاص \lr{\mbox{[2.1, 4.8]}} باشد.»}

\end{PSquote}

\textbf{چرا نادرست است:} \lr{$\mu$} یک عدد ثابت (غیرتصادفی) است. یا در \lr{\mbox{[2.1, 4.8]}} است یا نیست \lr{—} هیچ احتمالی درباره آن وجود ندارد. \textbf{بازه} تصادفی است (از نمونه‌ای به نمونه دیگر تغییر می‌کند)، نه پارامتر.

\PSsubsection{قیاس}

بازه‌های اطمینان را مانند یک تور ماهیگیری در نظر بگیرید:

\begin{itemize}
\tightlist
\item
  پیش از پرتاب: شانس 95\% که تور شما ماهی را بگیرد (این رویه 95\% مواقع کار می‌کند)
\item
  پس از پرتاب: ماهی یا گرفته شده یا نشده \lr{—} دیگر احتمالی وجود ندارد
\item
  ماهی \lr{$(\mu)$} حرکت نمی‌کند؛ تور شما (بازه اطمینان) حرکت می‌کند
\end{itemize}

\PSsection{14.3}{بازه اطمینان برای میانگین \lr{—} واریانس معلوم (بازه \lr{$Z$})}

\PSsubsection{صورت‌بندی}

داده: \lr{$X_{1},\, \ldots,\, X_{n}$} مستقل و هم‌توزیع از \lr{$N(\mu,\, \sigma^{2})$} با \lr{$\sigma^{2}$} \textbf{معلوم}.

\PSsubsection{استخراج}

\lr{$\bar{X} \sim N\left(\mu,\, \frac{\sigma^{2}}{n}\right)$}، بنابراین \lr{$Z = \frac{\bar{X} - \mu}{\frac{\sigma}{\sqrt{n}}} \sim N(0,\,1)$}

\lr{$P(-z_{\alpha /2} \le Z \le z_{\alpha /2}) = 1 - \alpha$}

با حل برای \lr{$\mu$}:\PSdisplay{P\left(\bar{X} - z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \leq \mu \leq \bar{X} + z_{\alpha/2}\frac{\sigma}{\sqrt{n}}\right) = 1 - \alpha}

\PSsubsection{فرمول}

\PSdisplay{\PSmtext{CI: } \bar{X} \pm z_{\alpha/2} \cdot \frac{\sigma}{\sqrt{n}}}

\PSsubsection{مقادیر رایج \lr{$z$}}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}ccc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{سطح اطمینان} & \PSth{\lr{$\alpha$}} & \PSth{\lr{$z_{\alpha /2}$}} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
90\% & \PSnum{0.10} & \PSnum{1.645} \\
95\% & \PSnum{0.05} & \PSnum{1.960} \\
99\% & \PSnum{0.01} & \PSnum{2.576} \\
\end{longtable}
}

\begin{PSbox}[unbreakable]{ex}{مثال}

اندازه‌گیری‌های نویز: \lr{$n = 36$}، \lr{$\bar{X} = 0.52\,\mathrm{V}$}، \lr{$\sigma = 0.12\,\mathrm{V}$} (معلوم از مشخصات تجهیز).

بازه اطمینان 95\%: \lr{$0.52 \pm 1.96 \times \frac{0.12}{\sqrt{36}} = 0.52 \pm 0.0392 = \ast \ast [0.481,\, 0.559] V$}**

\end{PSbox}

\PSsection{14.4}{بازه اطمینان برای میانگین \lr{—} واریانس نامعلوم (بازه \lr{$t$})}

\begin{PSbox}[unbreakable]{prob}{مسیله}

معمولاً \lr{$\sigma$} نامعلوم است. آن را با \lr{$S$} برآورد می‌کنیم. اما \lr{$\frac{\bar{X}}{\frac{S}{\sqrt{n}}}$} از \lr{$N(0,\,1)$} پیروی نمی‌کند.

\end{PSbox}

\PSsubsection{توزیع \lr{$t$} استیودنت}

\PSdisplay{T = \frac{\bar{X} - \mu}{S/\sqrt{n}} \sim t(n-1)}

توزیع \lr{$t$}:

\begin{itemize}
\tightlist
\item
  متقارن حول 0
\item
  دنباله‌های سنگین‌تر از \lr{$N(0,\,1)$} (عدم‌قطعیت بیشتر چون \lr{$S$} برآورد شده است)
\item
  درجات آزادی: \lr{$\nu = n - 1$}
\item
  وقتی \lr{\mbox{$\nu \to \infty$: $t(\nu) \to N(0,\,1)$}}
\item
  برای \lr{$\nu \ge 30$}: تقریباً \lr{$t \approx z$}
\end{itemize}

\PSsubsection{فرمول}

\PSdisplay{\PSmtext{CI: } \bar{X} \pm t_{\alpha/2, n-1} \cdot \frac{S}{\sqrt{n}}}

\begin{PSbox}[unbreakable]{ex}{مثال}

تأخیر قطعه: \lr{$n = 15$}، \lr{\mbox{$\bar{X} = 2.34$ ns}}، \lr{\mbox{$S = 0.45$ ns}}.

بازه اطمینان 95\%: \lr{$t_{0.025, 14} = 2.145$}

\lr{\mbox{CI}}: \lr{$2.34 \pm 2.145 \times \frac{0.45}{\sqrt{15}} = 2.34 \pm 0.249 = \ast \ast [2.09,\, 2.59]$ ns}**

(پهن‌تر از بازه \lr{$Z$} به‌دلیل عدم‌قطعیت اضافی در \lr{\mbox{S}}.)

\end{PSbox}

\PSsection{14.5}{بازه اطمینان برای واریانس}

\PSsubsection{صورت‌بندی}

داده از \lr{$N(\mu,\, \sigma^{2})$}. بازه اطمینان برای \lr{$\sigma^{2}$} مطلوب است.

\PSsubsection{توزیع کلیدی}

\PSdisplay{(n-1)S^2/\sigma^2 \sim \chi^2(n-1)}

\PSsubsection{فرمول}

\PSdisplay{\left[\frac{(n-1)S^2}{\chi^2_{\alpha/2, n-1}}, \quad \frac{(n-1)S^2}{\chi^2_{1-\alpha/2, n-1}}\right]}

\begin{PSbox}[unbreakable]{ex}{مثال}

توان نویز: \lr{$n = 20$}، \lr{$S^{2} = 0.045\,\mathrm{V}^{2}$}.

بازه اطمینان 95\% برای \lr{\mbox{$\sigma^{2}$: $\chi^{2}_{0.025, 19} = 32.852$}}، \lr{$\chi^{2}_{0.975, 19} = 8.907$}

\lr{\mbox{CI}}: \lr{$\left[\frac{(19)(0.045)}{32.852\text{\rl{،}}} \frac{(19)(0.045)}{8.907}\right] = [0.026,\, 0.096] V^{2}$}

بازه اطمینان 95\% برای \lr{$\sigma$: $\left[\sqrt{0.026},\, \sqrt{0.096}\right] = [0.161,\, 0.310] V$}

\end{PSbox}

\PSsection{14.6}{بازه اطمینان برای نسبت‌ها}

\PSsubsection{صورت‌بندی}

\lr{$n$} آزمایش، \lr{$\hat{x} = \frac{k}{n}$} موفقیت. برآورد \lr{$p$} = نسبت واقعی.

\PSsubsection{فرمول نمونه بزرگ (بازه والد)}

\PSdisplay{\hat{p} \pm z_{\alpha/2}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}}

\begin{PSbox}[unbreakable]{ex}{مثال: نرخ خطای بیت}

\PSnum{100,000} بیت ارسال شد، 53 خطا مشاهده شد. \lr{$\hat{p} = \frac{53}{100000} = 5.3 \times 10^{-4}$}.

بازه اطمینان 95\%: \lr{$5.3 \times 10^{-4} \pm 1.96\sqrt{5.3 \times 10^{-4} \times \frac{0.9995}{100000}} = 5.3 \times 10^{-4} \pm 1.43 \times 10^{-4}$}

\lr{\mbox{CI}}: \lr{\mbox{\textbf{\lr{$[3.9 \times 10^{-4},\, 6.7 \times 10^{-4}]$}}}}

\end{PSbox}

\PSsection{14.8}{تعیین اندازه نمونه}

\PSsubsection{پرسش: چند اندازه‌گیری لازم دارم؟}

برای پهنای بازه اطمینان \lr{$\le 2E$} (حاشیه خطای \lr{$E$}):

\PSdisplay{n \geq \left(\frac{z_{\alpha/2} \cdot \sigma}{E}\right)^2}

\begin{PSbox}[unbreakable]{ex}{مثال}

می‌خواهیم میانگین تأخیر را با اطمینان 95\% و در محدوده \lr{$\pm 0.5\,\mathrm{ms}$} برآورد کنیم. انتظار \lr{$\sigma \approx 3\,\mathrm{ms}$}.

\lr{$n \ge \left(1.96 \times \frac{3}{0.5}\right)^{2} = (11.76)^{2} = 138.3$ \ensuremath{\to}} به \textbf{\lr{$n = 139$} اندازه‌گیری} نیاز است

\end{PSbox}

\PSsection{14.10}{مسایل تمرینی}

یک حسگر ولتاژ را اندازه‌گیری می‌کند. از 50 اندازه‌گیری: \lr{$\bar{X} = 3.32\,\mathrm{V}$}، \lr{$S = 0.15\,\mathrm{V}$}.

\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  بازه اطمینان 95\% برای \lr{$\mu$} بسازید. (ب) بازه اطمینان 99\% برای \lr{$\mu$} بسازید. (ج) کدام پهن‌تر است و چرا؟

\begin{enumerate}

\item
  \lr{\mbox{$t_{0.025,49} \approx 2.010$. CI}}: \lr{$3.32 \pm 2.010\left(\frac{0.15}{\sqrt{50}}\right) = 3.32 \pm 0.043 = [3.277,\, 3.363]$}
\item
