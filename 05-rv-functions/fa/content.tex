\PSsection{5.2}{تبدیل‌های خطی: \lr{\mbox{$Y$ = aX $+ b$}}}

\PSsubsection{نتیجه}

اگر \lr{$X$} دارای \lr{\mbox{PDF $f_{X}(x)$}} باشد، آنگاه \lr{\mbox{$Y$ = aX $+ b$}} دارای:

\PSdisplay{f_Y(y) = \frac{1}{|a|} f_X\left(\frac{y-b}{a}\right)}

\PSsubsection{میانگین و واریانس}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{$E[Y]$ = aE[X] $+ b$}}
\item
  \lr{$\operatorname{Var}(Y) = a^{2}\operatorname{Var}(X)$}
\end{itemize}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: تقویت‌کننده}

سیگنال ورودی \lr{$X \sim N(0,\, \sigma^{2})$} از تقویت‌کننده‌ای با بهره \lr{$G = 10$} و آفست \lr{\mbox{DC}} یعنی \lr{$V_{0} = 1.5\,\mathrm{V}$} می‌گذرد.

خروجی: \lr{$Y = 10X + 1.5$}

\begin{itemize}
\tightlist
\item
  \lr{$Y \sim N(10 \cdot 0 + 1.5,\, 10^{2} \cdot \sigma^{2}) = N(1.5,\, 100\sigma^{2})$}
\item
  توان نویز خروجی به اندازه \lr{$G^{2} = 100$} افزایش یافته است
\end{itemize}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: حسگر دما}

خروجی حسگر \lr{$V = 0.01 \cdot T + 0.5$} (\lr{\mbox{V}} بر حسب ولت، \lr{$T$} بر حسب \lr{\mbox{\ensuremath{^{\circ}}C}})

اگر \lr{$T \sim N(25,\, 4)$: $V \sim N(0.01 \times 25 + 0.5,\, 0.01^{2} \times 4) = N(0.75,\, 0.0004)$}

\end{PSbox}

\PSsection{5.3}{تبدیل‌های یکنوای عمومی}

\PSsubsection{روش \lr{\mbox{CDF}} (کلی‌ترین روش)}

برای \lr{$Y = g(X)$}، \lr{$f_{Y}(y)$} را به این ترتیب بیابید:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  بنویسید \lr{$F_{Y}(y) = P(Y \le y) = P(g(X) \le y)$}
\item
  نامساوی را برای \lr{$X$} حل کنید
\item
  بر حسب \lr{$F_{X}$} بیان کنید
\item
  مشتق بگیرید تا \lr{$f_{Y}(y)$} به دست آید
\end{enumerate}

\PSsubsection{فرمول برای \lr{$g$} یکنوا}

اگر \lr{$g$} اکیداً صعودی با وارون \lr{$g^{-1}$} باشد:\PSdisplay{f_Y(y) = f_X(g^{-1}(y)) \cdot \frac{d}{dy}[g^{-1}(y)]}

اگر \lr{$g$} اکیداً نزولی باشد:\PSdisplay{f_Y(y) = f_X(g^{-1}(y)) \cdot \left|\frac{d}{dy}[g^{-1}(y)]\right|}

ترکیبی (برای هر دو کار می‌کند):\PSdisplay{f_Y(y) = \frac{f_X(x)}{|g'(x)|}\bigg|_{x=g^{-1}(y)}}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: \lr{\mbox{SNR}} بر حسب \lr{\mbox{dB}}}

\lr{\mbox{SNR}} خطی: \lr{$X \sim \mathrm{Exponential}(1)$} (توان محوشدگی رایلی نرمال‌شده).\PSbr{}\lr{\mbox{SNR}} بر حسب \lr{\mbox{dB}}: \lr{$Y = 10 \cdot \log_{10}(X)$}

\begin{itemize}
\tightlist
\item
  \lr{$g(x) = 10 \cdot \log_{10}(x)$}، \lr{$g'(x) = \frac{10}{x \cdot \ln 10}$}
\item
  \lr{$g^{-1}(y) = 10^{y/10}$}
\end{itemize}

\PSdisplay{f_Y(y) = \frac{f_X(10^{y/10})}{10/(10^{y/10} \cdot \ln 10)} = \frac{\ln 10}{10} \cdot 10^{y/10} \cdot e^{-10^{y/10}}}

این توزیع گاوسی نیست \lr{—} دنباله چپ بلندتری دارد (فروافت‌های عمیق).

\end{PSbox}

\PSsection{5.4}{تبدیل‌های غیرخطی: حالت غیریکنوا}

\PSsubsection{هنگامی که \lr{$Y = g(X)$} یک‌به‌یک نیست}

اگر چند مقدار \lr{$x$} به همان \lr{$y$} نگاشت شوند، روی تمام جواب‌ها جمع کنید:

\PSdisplay{f_Y(y) = \sum_{i} \frac{f_X(x_i)}{|g'(x_i)|}}

که در آن \lr{$x_{1},\, x_{2}$}, \lr{…} تمام ریشه‌های \lr{$g(x) = y$} هستند.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: توان از ولتاژ \lr{$(Y = X^{2})$}}

ولتاژ \lr{$X \sim N(0,\, \sigma^{2})$}. توان: \lr{$Y = X^{2}$}.

برای \lr{$y > 0$}، جواب‌ها \lr{$x = +\sqrt{y}$} و \lr{$x = -\sqrt{y}$} هستند. \lr{$g'(x) = 2x$}.

\PSdisplay{f_Y(y) = \frac{f_X(\sqrt{y})}{2\sqrt{y}} + \frac{f_X(-\sqrt{y})}{2\sqrt{y}} = \frac{1}{\sqrt{2\pi}\sigma} \cdot \frac{e^{-y/(2\sigma^2)}}{\sqrt{y}}}

این یک \textbf{توزیع کای‌دو با 1 درجه آزادی} است (مقیاس‌شده با \lr{$\sigma^{2}$}).

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: یکسوساز تمام‌موج \lr{$(Y = \lvert X\rvert)$}}

ورودی \lr{$X \sim N(0,\, \sigma^{2})$}. خروجی \lr{$Y = \lvert X\rvert$}.

برای \lr{\mbox{$y > 0$: $x = +y$}} و \lr{$x = -y$} جواب هستند. \lr{$\lvert g'(x)\rvert = 1$}.

\PSdisplay{f_Y(y) = f_X(y) + f_X(-y) = \frac{2}{\sigma\sqrt{2\pi}} e^{-y^2/(2\sigma^2)}, \quad y \geq 0}

این توزیع \textbf{نرمال تاشده} (نیم‌نرمال) است.

\end{PSbox}

\PSsection{5.5}{روش ژاکوبی}

\PSsubsection{رویه برای \lr{$Y = g(X)$}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  تبدیل \lr{$y = g(x)$} را مشخص کنید
\item
  وارون را بیابید: \lr{$x = g^{-1}(y)$}
\item
  ژاکوبی را محاسبه کنید: \lr{$J = \lvert \frac{dx}{dy}\rvert = \lvert \frac{d[g^{-1}(y)]}{dy}\rvert$}
\item
  اعمال کنید: \lr{$f_{Y}(y) = f_{X}(g^{-1}(y)) \cdot \lvert J\rvert$}
\end{enumerate}

\begin{PSbox}[unbreakable]{ex}{مثال گام‌به‌گام: نمایی گاوسی}

\lr{\mbox{$X \sim N(\mu,\, \sigma^{2})$. $Y = e^{x}$}} (تبدیل لگ‌نرمال).

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{$g(x) = e^{x}$} (یکنوای صعودی)
\item
  \lr{$x = \ln (y)$}، معتبر برای \lr{$y > 0$}
\item
  \lr{$\displaystyle J = \lvert \frac{dx}{dy}\rvert = \frac{1}{y}$}
\item
  \lr{$\displaystyle f_{Y}(y) = f_{X}(\ln y) \cdot \frac{1}{y} = \frac{1}{y\sigma \sqrt{2\pi}} \cdot \exp \left(-\frac{(\ln y - \mu)^{2}}{2\sigma^{2}}\right)$}
\end{enumerate}

این همان \textbf{توزیع لگ‌نرمال} است \lr{—} بسیاری از کمیت‌های مهندسی را مدل می‌کند (سایه‌اندازی در بی‌سیم، قیمت سهام، عمر قطعات همراه با فرسودگی).

\end{PSbox}

\PSsection{5.6}{کاربرد مهندسی: رایلی از گاوسی}

\PSsubsection{استخراج توزیع رایلی}

در مخابرات، سیگنال دریافتی مؤلفه‌های هم‌فاز \lr{$(I)$} و مربعی \lr{$(Q)$} دارد:

\begin{itemize}
\tightlist
\item
  \lr{$I \sim N(0,\, \sigma^{2})$}، \lr{$Q \sim N(0,\, \sigma^{2})$}، مستقل
\end{itemize}

پوش: \lr{$R = \sqrt{I^{2} + Q^{2}}$}

با استفاده از تبدیل \lr{$(I,\, Q) \to (R,\, \theta)$} در مختصات قطبی:

\begin{itemize}
\tightlist
\item
  \lr{$I = R \cdot \cos (\theta)$}، \lr{$Q = R \cdot \sin (\theta)$}
\item
  ژاکوبی: \lr{$\lvert \frac{\partial (I,\,Q)}{\partial (R,\,\theta)}\rvert = R$}
\end{itemize}

\lr{\mbox{PDF}} مشترک \lr{$(R,\, \theta)$}:\PSbr{}\lr{$\displaystyle f_{R,\Theta}(r,\,\theta) = f_{I,Q}(r \cdot \cos \theta,\, r \cdot \sin \theta) \cdot r = \frac{r}{2\pi \sigma^{2}} \cdot e^{-r^{2}/(2\sigma^{2})}$}

حاشیه‌سازی روی \lr{$\theta \in [0,\, 2\pi)$}:

\PSdisplay{f_R(r) = \frac{r}{\sigma^2} e^{-r^2/(2\sigma^2)}, \quad r \geq 0}

این همان \textbf{توزیع رایلی} است \lr{—} بنیادی برای مدل‌سازی کانال محوشدگی بی‌سیم.

\PSsection{5.8}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

سیگنال \lr{$X \sim \mathrm{Uniform}[0,\, 1]$}. خروجی \lr{\mbox{$Y = -2 \cdot \ln (X)$. PDF}} مربوط به \lr{$Y$} را بیابید و توزیع را تشخیص دهید.

\PSsolution{} روش \lr{\mbox{CDF}}: \lr{$F_{Y}(y) = P(-2\ln (X) \le y) = P(X \ge e^{-y/2}) = 1 - e^{-y/2}$} برای \lr{$y \ge 0$}.\PSbr{}\lr{$f_{Y}(y) = \frac{1}{2}e^{-y/2} \to Y \sim \mathrm{Exponential}(\beta = 2)$}. (این همان روش \lr{\mbox{CDF}} وارون برای تولید توزیع نمایی است!)

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

ولتاژ نویز \lr{$X \sim N(0,\, \sigma^{2})$} با \lr{$\sigma = 2\,\mathrm{V}$}. توان تلف‌شده در \lr{\mbox{$R = 50\,\Omega$: $P = \frac{X^{2}}{R}$}}.

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{$E[P]$} را بیابید. (ب) \lr{\mbox{PDF}} مربوط به \lr{$P$} را بیابید. (ج) \lr{$P(P > 0.2\,\mathrm{W})$} را بیابید.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{$\displaystyle E[P] = \frac{E[X^{2}]}{R} = \frac{\sigma^{2}}{R} = \frac{4}{50} = 0.08\,\mathrm{W}$}
\item
  \lr{$P = \frac{X^{2}}{50}$}. فرض کنید \lr{$W = X^{2} \sim \sigma^{2} \cdot \chi^{2}(1)$}. آنگاه \lr{$P = \frac{W}{50}$. $f_{P}(p) = 50 \cdot f_{W}(50p) = \frac{50}{2\sigma^{2}} \cdot \frac{50p}{\sigma^{2}}^{-1/2} \cdot e^{-50p/(2\sigma^{2})}$} برای \lr{$p > 0$}.
\item
  از \lr{\mbox{MATLAB}} استفاده کنید: \lr{\PScode{1 - chi2cdf(0.2*50/4, 1)} $= 1 - \texttt{chi2cdf}(2.5,\, 1) \approx 0.114$}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 3}

\lr{\mbox{$X \sim \mathrm{Exponential}(\lambda = 1)$. $Y = \sqrt{X}$}}. مقدار \lr{$f_{Y}(y)$} را بیابید.

\PSsolution{} \lr{$g(x) = \sqrt{x} \to x = y^{2}$}، \lr{$\frac{dx}{dy} = 2y$}.\PSbr{}\lr{$f_{Y}(y) = f_{X}(y^{2}) \cdot \lvert 2y\rvert = e^{-y^{2}} \cdot 2y$} برای \lr{$y \ge 0$}. این یک توزیع رایلی با \lr{$\sigma^{2} = \frac{1}{2}$} است.

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 4}

فاصله \lr{\mbox{$D \sim \mathrm{Uniform}[1,\, 10]$ km}}. تلفات مسیر: \lr{\mbox{$L = 20 \cdot \log_{10}(D)$ dB}}. مقدار \lr{$E[L]$} و \lr{\mbox{PDF}} مربوط به \lr{$L$} را بیابید.

\PSsolution{} \lr{$\displaystyle E[L] = E[20 \cdot \log_{10}(D)] = \int_{1}^{10} 20 \cdot \log_{10}(d) \cdot \frac{1}{9}\,dd = \frac{20}{9} \cdot \int_{1}^{10} \log_{10}(d)\,dd$}\PSbr{}\lr{$= \frac{20}{9} \cdot \left[d \cdot \log_{10}(d) - \frac{d}{\ln (10)}\right]_{1}^{10} = \frac{20}{9} \cdot \left[10 - \frac{10}{\ln 10} + \frac{1}{\ln 10}\right] = \frac{20}{9} \cdot \left(10 - \frac{9}{\ln 10}\right) \approx 13.55$ dB}

\lr{\mbox{PDF}}: \lr{$L$} از 0 تا \lr{\mbox{20 dB}} تغییر می‌کند. \lr{$g^{-1}(l) = 10^{l/20}$}، \lr{$\lvert \frac{dg^{-1}}{dl}\rvert = \frac{\ln 10}{20} \cdot 10^{l/20}$}.\PSbr{}\lr{$f_{L}(l) = \frac{1}{9} \cdot \frac{\ln 10}{20} \cdot 10^{l/20}$} برای \lr{$0 \le l \le 20$}.

\end{PSbox}
